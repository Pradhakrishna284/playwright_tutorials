// fixtures/multiFixtures.ts
import { test as base } from '@playwright/test';

type MultiFixtures = {
  config: AppConfig;
  apiClient: ApiClient;
  database: Database;
};

type AppConfig = {
  apiUrl: string;
  dbUrl: string;
};

class ApiClient {
  constructor(public url: string) {}
  async get(path: string) {
    const response = await fetch(`${this.url}${path}`);
    return response.json();
  }
}

class Database {
  static async connect(url: string) {
    console.log(`Connected to ${url}`);
    return new Database(url);
  }
  
  constructor(public url: string) {}
  
  async query(sql: string) {
    console.log(`Executing: ${sql}`);
    return [];
  }
  
  async close() {
    console.log('Database connection closed');
  }
}

export const test = base.extend<MultiFixtures>({
  // First fixture: Configuration
  config: async ({}, use) => {
    const config = {
      apiUrl: process.env.API_URL || 'http://localhost:3000',
      dbUrl: process.env.DB_URL || 'postgres://localhost/test',
    };
    await use(config);
  },
  
  // Second fixture: Uses config
  apiClient: async ({ config }, use) => {
    const client = new ApiClient(config.apiUrl);
    console.log(`API Client initialized: ${config.apiUrl}`);
    await use(client);
  },
  
  // Third fixture: Uses config
  database: async ({ config }, use) => {
    const db = await Database.connect(config.dbUrl);
    await use(db);
    await db.close();
  },
});

Example 4: Test-Scoped vs Worker-Scoped Comparison
---------------------------------------------------

// fixtures/scopeComparison.ts
import { test as base } from '@playwright/test';

type ScopeFixtures = {
  testScopedCounter: { count: number; increment: () => void };
  workerScopedCounter: { count: number; increment: () => void };
};

export const test = base.extend<ScopeFixtures>({
  // TEST-SCOPED: Fresh for each test
  testScopedCounter: async ({}, use) => {
    const counter = {
      count: 0,
      increment() {
        this.count++;
      }
    };
    console.log('Test-scoped counter created');
    await use(counter);
    console.log(`Test-scoped counter final value: ${counter.count}`);
  },
  
  // WORKER-SCOPED: Shared across tests in worker
  workerScopedCounter: [async ({}, use) => {
    const counter = {
      count: 0,
      increment() {
        this.count++;
      }
    };
    console.log('Worker-scoped counter created');
    await use(counter);
    console.log(`Worker-scoped counter final value: ${counter.count}`);
  }, { scope: 'worker' }],
});

export { expect } from '@playwright/test';

// fixtures/autoFixtures.ts
import { test as base } from '@playwright/test';

export const test = base.extend({
  // Runs automatically for every test
  performanceMonitor: [async ({ page }, use) => {
    // Setup: Start monitoring
    const metrics = [];
    
    page.on('metrics', data => {
      metrics.push({
        timestamp: Date.now(),
        metrics: data
      });
    });
    
    console.log('📊 Performance monitoring started');
    
    // No need to provide anything
    await use();
    
    // Teardown: Log summary
    console.log(`📈 Collected ${metrics.length} metric samples`);
    
  }, { auto: true }],
  
  // Another auto fixture: Screenshot on failure
  screenshotOnFailure: [async ({ page }, use, testInfo) => {
    await use();
    
    // Runs after test
    if (testInfo.status !== testInfo.expectedStatus) {
      const screenshot = await page.screenshot();
      await testInfo.attach('failure-screenshot', {
        body: screenshot,
        contentType: 'image/png'
      });
      console.log('📸 Screenshot captured on failure');
    }
  }, { auto: true }],
});

export { expect } from '@playwright/test';

// Usage
import { test, expect } from '../fixtures/autoFixtures';

test('normal test', async ({ page }) => {
  // performanceMonitor and screenshotOnFailure run automatically!
  await page.goto('https://example.com');
  await expect(page.locator('h1')).toBeVisible();
});

// Output:
// 📊 Performance monitoring started
// 📈 Collected 5 metric samples

// fixtures/apiAuthFixtures.ts
import { test as base } from '@playwright/test';

type ApiAuthFixtures = {
  apiToken: string;
  authenticatedRequest: {
    get: (url: string) => Promise<any>;
    post: (url: string, data: any) => Promise<any>;
    put: (url: string, data: any) => Promise<any>;
    delete: (url: string) => Promise<any>;
  };
};

export const test = base.extend<ApiAuthFixtures>({
  apiToken: async ({ request }, use) => {
    // Get authentication token
    const response = await request.post('https://api.example.com/auth/token', {
      data: {
        username: process.env.API_USERNAME || 'testuser',
        password: process.env.API_PASSWORD || 'password123'
      }
    });
    
    const { token } = await response.json();
    console.log('🔑 API token obtained');
    
    await use(token);
  },
  
  authenticatedRequest: async ({ request, apiToken }, use) => {
    const authHeaders = {
      'Authorization': `Bearer ${apiToken}`,
      'Content-Type': 'application/json'
    };
    
    const authenticatedRequest = {
      get: async (url: string) => {
        const response = await request.get(url, { headers: authHeaders });
        return response.json();
      },
      
      post: async (url: string, data: any) => {
        const response = await request.post(url, {
          data,
          headers: authHeaders
        });
        return response.json();
      },
      
      put: async (url: string, data: any) => {
        const response = await request.put(url, {
          data,
          headers: authHeaders
        });
        return response.json();
      },
      
      delete: async (url: string) => {
        const response = await request.delete(url, { headers: authHeaders });
        return response.ok();
      },
    };
    
    await use(authenticatedRequest);
  },
});

// fixtures/databaseFixtures.ts
import { test as base } from '@playwright/test';
import { Pool } from 'pg'; // PostgreSQL example

type DatabaseFixtures = {
  db: {
    query: (sql: string, params?: any[]) => Promise<any[]>;
    insert: (table: string, data: object) => Promise<any>;
    update: (table: string, id: number, data: object) => Promise<any>;
    delete: (table: string, id: number) => Promise<void>;
  };
};

export const test = base.extend<DatabaseFixtures>({
  db: [async ({}, use) => {
    // Connect to database (once per worker)
    const pool = new Pool({
      host: process.env.DB_HOST || 'localhost',
      database: process.env.DB_NAME || 'test_db',
      user: process.env.DB_USER || 'test_user',
      password: process.env.DB_PASSWORD || 'password',
      port: parseInt(process.env.DB_PORT || '5432'),
    });
    
    console.log('🗄️  Database connection established');
    
    const db = {
      query: async (sql: string, params?: any[]) => {
        const result = await pool.query(sql, params);
        return result.rows;
      },
      
      insert: async (table: string, data: object) => {
        const keys = Object.keys(data);
        const values = Object.values(data);
        const placeholders = keys.map((_, i) => `$${i + 1}`).join(', ');
        
        const sql = `INSERT INTO ${table} (${keys.join(', ')}) VALUES (${placeholders}) RETURNING *`;
        const result = await pool.query(sql, values);
        return result.rows[0];
      },
      
      update: async (table: string, id: number, data: object) => {
        const keys = Object.keys(data);
        const values = Object.values(data);
        const setClause = keys.map((key, i) => `${key} = $${i + 1}`).join(', ');
        
        const sql = `UPDATE ${table} SET ${setClause} WHERE id = $${keys.length + 1} RETURNING *`;
        const result = await pool.query(sql, [...values, id]);
        return result.rows[0];
      },
      
      delete: async (table: string, id: number) => {
        await pool.query(`DELETE FROM ${table} WHERE id = $1`, [id]);
      },
    };
    
    await use(db);
    
    // Close connection pool
    await pool.end();
    console.log('🗄️  Database connection closed');
    
  }, { scope: 'worker' }],
});
