EXAMPLE 1: Complete E-Commerce Homepage Visual Testing
-------------------------------------------------------

```javascript
// tests/visual/homepage.spec.ts
import { test, expect } from '@playwright/test';

test.describe('Homepage Visual Regression', () => {
  
  test('full homepage layout', async ({ page }) => {
    await page.goto('/');
    
    // Wait for all images to load
    await page.waitForLoadState('networkidle');
    
    // Disable animations for consistency
    await page.addStyleTag({
      content: `
        *, *::before, *::after {
          animation-duration: 0s !important;
          animation-delay: 0s !important;
          transition-duration: 0s !important;
          transition-delay: 0s !important;
        }
      `
    });
    
    await expect(page).toHaveScreenshot('homepage-full.png', {
      fullPage: true,
      animations: 'disabled',
      mask: [
        page.locator('.timestamp'),
        page.locator('.live-counter'),
        page.locator('.advertisement'),
      ]
    });
  });
  
  test('hero section', async ({ page }) => {
    await page.goto('/');
    await expect(page.locator('.hero-section')).toHaveScreenshot('hero.png');
  });
  
  test('navigation menu', async ({ page }) => {
    await page.goto('/');
    await expect(page.locator('nav')).toHaveScreenshot('navigation.png');
  });
  
  test('featured products grid', async ({ page }) => {
    await page.goto('/');
    await expect(page.locator('.featured-products')).toHaveScreenshot('featured-products.png');
  });
  
  test('footer', async ({ page }) => {
    await page.goto('/');
    await expect(page.locator('footer')).toHaveScreenshot('footer.png');
  });
});
```

Why This Works:
• Full page + component tests for comprehensive coverage
• Animations disabled for consistency
• Dynamic content masked
• Network idle ensures all images loaded
• Clear, descriptive screenshot names

EXAMPLE 2: Responsive Design Visual Testing
--------------------------------------------

```javascript
// tests/visual/responsive.spec.ts
import { test, expect, devices } from '@playwright/test';

const viewports = [
  { name: 'mobile-portrait', width: 375, height: 667 },
  { name: 'mobile-landscape', width: 667, height: 375 },
  { name: 'tablet-portrait', width: 768, height: 1024 },
  { name: 'tablet-landscape', width: 1024, height: 768 },
  { name: 'desktop-hd', width: 1920, height: 1080 },
  { name: 'desktop-4k', width: 3840, height: 2160 },
];

const pages = ['/', '/products', '/about', '/contact'];

test.describe('Responsive Visual Regression', () => {
  for (const viewport of viewports) {
    test.describe(viewport.name, () => {
      test.use({ 
        viewport: { width: viewport.width, height: viewport.height } 
      });
      
      for (const pagePath of pages) {
        test(`${pagePath} on ${viewport.name}`, async ({ page }) => {
          await page.goto(pagePath);
          await page.waitForLoadState('networkidle');
          
          const screenshotName = `${pagePath.replace('/', 'home')}-${viewport.name}.png`;
          
          await expect(page).toHaveScreenshot(screenshotName, {
            fullPage: true,
            animations: 'disabled',
          });
        });
      }
    });
  }
});
```

```
home-tablet-landscape.png
products-desktop-hd.png
... (24 total screenshots: 4 pages × 6 viewports)
```

EXAMPLE 3: Component Library Visual Testing
--------------------------------------------

```javascript
// tests/visual/components.spec.ts
import { test, expect } from '@playwright/test';

const components = [
  { name: 'button-primary', selector: '.btn-primary' },
  { name: 'button-secondary', selector: '.btn-secondary' },
  { name: 'button-danger', selector: '.btn-danger' },
  { name: 'card', selector: '.card' },
  { name: 'modal', selector: '.modal' },
  { name: 'input-text', selector: 'input[type="text"]' },
  { name: 'input-email', selector: 'input[type="email"]' },
  { name: 'checkbox', selector: 'input[type="checkbox"]' },
  { name: 'radio', selector: 'input[type="radio"]' },
  { name: 'select', selector: 'select' },
  { name: 'alert-success', selector: '.alert-success' },
  { name: 'alert-error', selector: '.alert-error' },
];

const states = ['default', 'hover', 'focus', 'active', 'disabled'];

test.describe('Component Visual Regression', () => {
  
  // Test each component in each state
  for (const component of components) {
    test.describe(component.name, () => {
      
      for (const state of states) {
        test(`${state} state`, async ({ page }) => {
          await page.goto('/components');
          
          const element = page.locator(component.selector).first();
          
          // Apply state
          switch (state) {
            case 'hover':
              await element.hover();
              break;
            case 'focus':
              await element.focus();
              break;
            case 'active':
              await element.click();
              await page.waitForTimeout(100);
              break;
            case 'disabled':
              await element.evaluate(el => el.setAttribute('disabled', 'true'));
              break;
          }
          
          await expect(element).toHaveScreenshot(`${component.name}-${state}.png`);
        });
      }
    });
  }
});
```

Output: 60 screenshots (12 components × 5 states)

Benefits:
✓ Validates design system consistency
✓ Catches CSS regressions in components
✓ Documents component states visually
✓ Easy to see impact of style changes

EXAMPLE 4: Dark Mode / Theme Testing
-------------------------------------

```javascript
// tests/visual/themes.spec.ts
import { test, expect } from '@playwright/test';

const themes = ['light', 'dark', 'high-contrast', 'blue', 'purple'];
const pages = ['/', '/dashboard', '/settings', '/profile'];

test.describe('Theme Visual Regression', () => {
  
  for (const theme of themes) {
    test.describe(`${theme} theme`, () => {
      
      test.beforeEach(async ({ page }) => {
        await page.goto('/');
        
        // Set theme
        await page.evaluate((themeName) => {
          localStorage.setItem('theme', themeName);
          document.documentElement.setAttribute('data-theme', themeName);
        }, theme);
      });
      
      for (const pagePath of pages) {
        test(`${pagePath}`, async ({ page }) => {
          await page.goto(pagePath);
          
          // Wait for theme to apply
          await page.waitForTimeout(100);
          await page.waitForLoadState('networkidle');
          
          await expect(page).toHaveScreenshot(`${pagePath.replace('/', 'home')}-${theme}.png`, {
            fullPage: true,
            animations: 'disabled',
          });
        });
      }
    });
  }
});
```

Output: 20 screenshots (5 themes × 4 pages)

EXAMPLE 8: Visual Testing with Fixtures (Advanced)
---------------------------------------------------

```javascript
// fixtures/screenshot.ts
import { test as base, expect } from '@playwright/test';

type ScreenshotFixtures = {
  visualTest: (name: string, options?: any) => Promise<void>;
};

export const test = base.extend<ScreenshotFixtures>({
  visualTest: async ({ page }, use) => {
    const takeScreenshot = async (name: string, options = {}) => {
      // Wait for fonts
      await page.evaluate(() => document.fonts.ready);
      
      // Wait for images
      await page.evaluate(async () => {
        const images = Array.from(document.images);
        await Promise.all(
          images
            .filter(img => !img.complete)
            .map(img => new Promise(resolve => {
              img.onload = img.onerror = resolve;
            }))
        );
      });
      
      // Disable animations
      await page.addStyleTag({
        content: `
          *, *::before, *::after {
            animation-duration: 0s !important;
            transition-duration: 0s !important;
          }
        `
      });
      
      await expect(page).toHaveScreenshot(name, {
        animations: 'disabled',
        ...options
      });
    };
    
    await use(takeScreenshot);
  },
});

export { expect } from '@playwright/test';
```

Usage:

```javascript
// tests/visual/with-fixture.spec.ts
import { test, expect } from '../fixtures/screenshot';

test('homepage with custom fixture', async ({ page, visualTest }) => {
  await page.goto('/');
  
  // Simplified screenshot with all optimizations
  await visualTest('homepage.png', {
    fullPage: true,
    mask: [page.locator('.timestamp')]
  });
});
```

EXAMPLE 10: CI/CD Visual Testing Workflow
------------------------------------------

```yaml
# .github/workflows/visual-tests.yml
name: Visual Regression Tests

on:
  push:
    branches: [main, develop]
  pull_request:
    branches: [main]

jobs:
  visual-tests:
    runs-on: ubuntu-latest
    timeout-minutes: 30
    
    steps:
      - name: Checkout code
        uses: actions/checkout@v3
      
      - name: Setup Node.js
        uses: actions/setup-node@v3
        with:
          node-version: 18
          cache: 'npm'
      
      - name: Install dependencies
        run: npm ci
      
      - name: Install Playwright browsers
        run: npx playwright install --with-deps
      
      - name: Run visual regression tests
        run: npx playwright test tests/visual/
      
      - name: Upload test results
        if: always()
        uses: actions/upload-artifact@v3
        with:
          name: visual-test-results
          path: |
            test-results/
            playwright-report/
          retention-days: 30
      
      - name: Upload screenshots on failure
        if: failure()
        uses: actions/upload-artifact@v3
        with:
          name: visual-diff-screenshots
          path: |
            test-results/**/*-actual.png
            test-results/**/*-expected.png
            test-results/**/*-diff.png
          retention-days: 30
      
      - name: Comment PR with results
        if: github.event_name == 'pull_request' && failure()
        uses: actions/github-script@v6
        with:
          script: |
            const fs = require('fs');
            const comment = `
            ## ⚠️ Visual Regression Tests Failed
            
            Visual differences detected. Please review the screenshots in the artifacts.
            
            [View Test Report](https://github.com/${{ github.repository }}/actions/runs/${{ github.run_id }})
            `;
            
            github.rest.issues.createComment({
              issue_number: context.issue.number,
              owner: context.repo.owner,
              repo: context.repo.repo,
              body: comment
            });
```

EXAMPLE 12: Visual Testing Performance Optimization
----------------------------------------------------

```javascript
// tests/visual/optimized.spec.ts
import { test, expect } from '@playwright/test';

// Optimize: Reuse browser context
test.describe.configure({ mode: 'serial' });

let sharedPage;

test.beforeAll(async ({ browser }) => {
  const context = await browser.newContext();
  sharedPage = await context.newPage();
  
  // Navigate once
  await sharedPage.goto('/');
  await sharedPage.waitForLoadState('networkidle');
  
  // Disable animations globally
  await sharedPage.addStyleTag({
    content: `
      *, *::before, *::after {
        animation: none !important;
        transition: none !important;
      }
    `
  });
});

test.afterAll(async () => {
  await sharedPage.close();
});

test('hero section', async () => {
  await expect(sharedPage.locator('.hero')).toHaveScreenshot('hero.png');
```

```javascript
// tests/performance/page-load.spec.ts
import { test, expect } from '@playwright/test';

test.describe('Page Load Performance', () => {
  
  test('homepage performance metrics', async ({ page }) => {
    const metrics = {
      timing: {},
      resources: [],
      webVitals: {},
    };
    
    // Start timer
    const navigationStart = Date.now();
    
    // Navigate to page
    await page.goto('/');
    
    // Get Navigation Timing
    metrics.timing = await page.evaluate(() => {
      const perf = performance.timing;
      return {
        dnsLookup: perf.domainLookupEnd - perf.domainLookupStart,
        tcpConnection: perf.connectEnd - perf.connectStart,
        serverResponse: perf.responseEnd - perf.requestStart,
        domInteractive: perf.domInteractive - perf.navigationStart,
        domComplete: perf.domComplete - perf.navigationStart,
        pageLoad: perf.loadEventEnd - perf.navigationStart,
      };
    });
    
    // Get Resource Timing
    metrics.resources = await page.evaluate(() => {
      return performance.getEntriesByType('resource').map(r => ({
        name: r.name,
        duration: r.duration,
        size: r.transferSize,
        type: r.initiatorType,
      }));
    });
    
    // Get LCP (Largest Contentful Paint)
    metrics.webVitals.lcp = await page.evaluate(() => {
      return new Promise(resolve => {
        new PerformanceObserver(list => {
          const entries = list.getEntries();
          const lastEntry = entries[entries.length - 1];
          resolve(lastEntry.renderTime || lastEntry.loadTime);
        }).observe({ entryTypes: ['largest-contentful-paint'] });
        
        setTimeout(() => resolve(0), 10000);
      });
    });
    
    // Get CLS (Cumulative Layout Shift)
    await page.waitForLoadState('networkidle');
    metrics.webVitals.cls = await page.evaluate(() => {
      return new Promise(resolve => {
        let clsScore = 0;
        
        new PerformanceObserver(list => {
          for (const entry of list.getEntries()) {
            if (!entry.hadRecentInput) {
              clsScore += entry.value;
            }
          }
          resolve(clsScore);
        }).observe({ entryTypes: ['layout-shift'] });
        
        setTimeout(() => resolve(clsScore), 5000);
      });
    });
    
    // Calculate total time
    metrics.totalTime = Date.now() - navigationStart;
    
    // Log all metrics
    console.log('Performance Metrics:');
    console.log('='.repeat(50));
    console.log('Timing:');
    console.log(`  DNS Lookup: ${metrics.timing.dnsLookup}ms`);
    console.log(`  TCP Connection: ${metrics.timing.tcpConnection}ms`);
    console.log(`  Server Response: ${metrics.timing.serverResponse}ms`);
    console.log(`  DOM Interactive: ${metrics.timing.domInteractive}ms`);
    console.log(`  DOM Complete: ${metrics.timing.domComplete}ms`);
    console.log(`  Page Load: ${metrics.timing.pageLoad}ms`);
    console.log(`  Total Time: ${metrics.totalTime}ms`);
    console.log();
    console.log('Resources:');
    console.log(`  Total: ${metrics.resources.length}`);
    console.log(`  JS: ${metrics.resources.filter(r => r.type === 'script').length}`);
    console.log(`  CSS: ${metrics.resources.filter(r => r.type === 'link').length}`);
    console.log(`  Images: ${metrics.resources.filter(r => r.type === 'img').length}`);
    console.log();
    console.log('Web Vitals:');
    console.log(`  LCP: ${metrics.webVitals.lcp}ms`);
    console.log(`  CLS: ${metrics.webVitals.cls}`);
    console.log('='.repeat(50));
    
    // Performance Budget Assertions
    expect(metrics.timing.pageLoad).toBeLessThan(3000); // 3 seconds
    expect(metrics.timing.domInteractive).toBeLessThan(1500); // 1.5 seconds
    expect(metrics.webVitals.lcp).toBeLessThan(2500); // 2.5 seconds
    expect(metrics.webVitals.cls).toBeLessThan(0.1); // 0.1 score
    expect(metrics.resources.length).toBeLessThan(50); // 50 resources
    
    // Find slow resources
    const slowResources = metrics.resources.filter(r => r.duration > 1000);
    expect(slowResources.length).toBe(0);
  });
});
```

```javascript
// tests/performance/api-performance.spec.ts
import { test, expect } from '@playwright/test';

test.describe('API Performance', () => {
  
  test('measure all API calls', async ({ page }) => {
    const apiCalls = [];
    
    page.on('response', response => {
      if (response.url().includes('/api/')) {
        const timing = response.timing();
        apiCalls.push({
          url: response.url(),
          method: response.request().method(),
          status: response.status(),
          duration: timing ? timing.responseEnd : 0,
          size: parseInt(response.headers()['content-length'] || '0'),
        });
      }
    });
    
    await page.goto('/dashboard');
    await page.waitForLoadState('networkidle');
    
    console.log(`Total API calls: ${apiCalls.length}`);
    console.log('='.repeat(80));
    
    apiCalls.forEach((api, index) => {
      console.log(`${index + 1}. ${api.method} ${api.url}`);
      console.log(`   Status: ${api.status} | Duration: ${api.duration}ms | Size: ${api.size} bytes`);
    });
    
    console.log('='.repeat(80));
    
    // Calculate statistics
    const avgDuration = apiCalls.reduce((sum, api) => sum + api.duration, 0) / apiCalls.length;
    const maxDuration = Math.max(...apiCalls.map(api => api.duration));
    const totalSize = apiCalls.reduce((sum, api) => sum + api.size, 0);
    
    console.log(`Average API Duration: ${avgDuration.toFixed(2)}ms`);
    console.log(`Max API Duration: ${maxDuration}ms`);
    console.log(`Total API Data: ${(totalSize / 1024).toFixed(2)} KB`);
    
    // Assertions
    expect(avgDuration).toBeLessThan(500); // 500ms average
    expect(maxDuration).toBeLessThan(2000); // 2 seconds max
    
    const slowAPIs = apiCalls.filter(api => api.duration > 1000);
    expect(slowAPIs.length).toBe(0);
  });
  
  test('parallel API calls performance', async ({ request }) => {
    const endpoints = [
      '/api/users',
      '/api/posts',
      '/api/comments',
      '/api/products',
      '/api/categories',
    ];
    
    const start = Date.now();
    
    // Call all APIs in parallel
    const promises = endpoints.map(endpoint => 
      request.get(`https://api.example.com${endpoint}`)
    );
    
    const responses = await Promise.all(promises);
    const totalDuration = Date.now() - start;
    
    console.log(`Total parallel execution time: ${totalDuration}ms`);
    
    responses.forEach((response, index) => {
      console.log(`${endpoints[index]}: ${response.status()}`);
    });
    
    // Should be faster than sequential
    expect(totalDuration).toBeLessThan(2000); // 2 seconds for all parallel
  });
});
```

```javascript
// tests/performance/resources.spec.ts
import { test, expect } from '@playwright/test';

test('measure page weight and resource count', async ({ page }) => {
  let totalSize = 0;
  let requestCount = 0;
  const resourcesByType = {
    js: { count: 0, size: 0 },
    css: { count: 0, size: 0 },
    images: { count: 0, size: 0 },
    fonts: { count: 0, size: 0 },
    other: { count: 0, size: 0 },
  };
  
  page.on('response', async response => {
    requestCount++;
    
    const size = parseInt(response.headers()['content-length'] || '0');
    totalSize += size;
    
    const contentType = response.headers()['content-type'] || '';
    
    if (contentType.includes('javascript')) {
      resourcesByType.js.count++;
      resourcesByType.js.size += size;
    } else if (contentType.includes('css')) {
      resourcesByType.css.count++;
      resourcesByType.css.size += size;
    } else if (contentType.includes('image')) {
      resourcesByType.images.count++;
      resourcesByType.images.size += size;
    } else if (contentType.includes('font')) {
      resourcesByType.fonts.count++;
      resourcesByType.fonts.size += size;
    } else {
      resourcesByType.other.count++;
      resourcesByType.other.size += size;
    }
  });
  
  await page.goto('/');
  await page.waitForLoadState('networkidle');
  
  console.log('Page Weight Analysis:');
  console.log('='.repeat(60));
  console.log(`Total Requests: ${requestCount}`);
  console.log(`Total Size: ${(totalSize / 1024 / 1024).toFixed(2)} MB`);
  console.log();
  console.log('By Resource Type:');
  Object.entries(resourcesByType).forEach(([type, data]) => {
    console.log(`  ${type.toUpperCase()}:`);
    console.log(`    Count: ${data.count}`);
    console.log(`    Size: ${(data.size / 1024).toFixed(2)} KB`);
  });
  console.log('='.repeat(60));
  
  // Performance budgets
  expect(totalSize).toBeLessThan(3 * 1024 * 1024); // 3MB total
  expect(resourcesByType.js.size).toBeLessThan(500 * 1024); // 500KB JS
  expect(resourcesByType.css.size).toBeLessThan(100 * 1024); // 100KB CSS
  expect(requestCount).toBeLessThan(50); // 50 requests
});
```
