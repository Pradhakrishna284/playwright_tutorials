TEST 1.1: Product Search and Filter
────────────────────────────────────────

import { test, expect } from '@playwright/test';

test('search product and apply filters', async ({ page }) => {
  // Navigate to e-commerce site
  await page.goto('https://demowebshop.tricentis.com');
  
  // FILL ACTION: Search for product
  const searchBox = page.getByPlaceholder('Search store');
  await searchBox.fill('laptop');
  
  // CLICK ACTION: Submit search
  const searchButton = page.getByRole('button', { name: /search/i });
  await searchButton.click();
  
  // Verify search results
  await expect(page.getByText(/laptop/i)).toBeVisible();
  
  // KEYBOARD ACTION: Clear and search again
  await searchBox.fill('');
  await searchBox.pressSequentially('phone', { delay: 50 });
  await searchButton.click();
  
  // Verify new results
  const results = page.locator('.product-item');
  expect(await results.count()).toBeGreaterThan(0);
});

test('inspect all available options before selection', async ({ page }) => {
  await page.goto('https://example.com/colors-form');
  
  const colorSelect = page.locator('#colors');
  
  // GET: All option elements
  const allOptions = page.locator('#colors > option');
  
  // GET: Total count
  const optionsCount = await allOptions.count();
  console.log(`Total options in dropdown: ${optionsCount}`);
  expect(optionsCount).toBe(7);
  
  // GET: All option texts
  const allOptionTexts = await allOptions.allTextContents();
  console.log('All option labels:', allOptionTexts);
  
  // GET: All option values
  const allOptionValues = await colorSelect.evaluate(() => {
    const options = document.querySelectorAll('option');
    return Array.from(options).map((option: any) => option.value);
  });
  console.log('All option values:', allOptionValues);
  
  // ITERATE: Through each option
  for (let i = 0; i < optionsCount; i++) {
    const optionLabel = await allOptions.nth(i).innerText();
    const optionValue = await allOptions.nth(i).getAttribute('value');
    console.log(`Option ${i + 1}: Label = ${optionLabel}, Value = ${optionValue}`);
  }
});

test('select options conditionally based on available options', async ({ page }) => {
  await page.goto('https://example.com/colors-form');
  
  const colorSelect = page.getByLabel('Choose Colors');
  
  // GET: Available options
  const availableOptions = await colorSelect.locator('option')
    .allTextContents();
  console.log('Available colors:', availableOptions);
  
  // SELECT: Only colors that exist (safe selection)
  const desiredColors = ['Red', 'Blue', 'Green'].filter(
    color => availableOptions.includes(color)
  );
  
  if (desiredColors.length > 0) {
    await colorSelect.selectOption(
      desiredColors.map(label => ({ label }))
    );
  }
  
  // VERIFY: Successfully selected
  const selectedCount = await colorSelect.evaluate(() => 
    document.querySelectorAll('option:checked').length
  );
  expect(selectedCount).toBe(desiredColors.length);
});

test('track selection count at different stages', async ({ page }) => {

  await colorSelect.selectOption(['blue', 'yellow']);

    document.querySelectorAll('option:checked').length
  );
  console.log(`Count after initial selection: ${selectedCount}`);

  // STAGE 2: Add more (now 4 total)

  selectedCount = await colorSelect.evaluate(() => 
    document.querySelectorAll('option:checked').length
  );
  console.log(`Count after adding more selections: ${selectedCount}`);
  expect(selectedCount).toBe(4);
  
  // STAGE 3: Partial re-selection (back to 2)
  await colorSelect.selectOption(['red', 'green']);
  selectedCount = await colorSelect.evaluate(() => 
    document.querySelectorAll('option:checked').length
  );
  console.log(`Count after partial reselection: ${selectedCount}`);
  expect(selectedCount).toBe(2);
  
  // STAGE 4: Complete deselection (0)
  await colorSelect.selectOption([]);
  selectedCount = await colorSelect.evaluate(() => 
    document.querySelectorAll('option:checked').length
  );
  console.log(`Count after complete deselection: ${selectedCount}`);
  expect(selectedCount).toBe(0);
});

  const shippingRadios = page.locator('input[name="shipping"]');
  
  // Check first option
  await shippingRadios.nth(0).check();
  await expect(shippingRadios.nth(0)).toBeChecked();
  
  // Switch to second option
  await shippingRadios.nth(1).check();
  await expect(shippingRadios.nth(1)).toBeChecked();

  // CHECKBOX: Multiple selections
  const notificationCheckboxes = page.locator(
    'input[type="checkbox"][name*="notification"]'
  );
  const count = await notificationCheckboxes.count();
  
  // Check first 3
  for (let i = 0; i < Math.min(3, count); i++) {
    await notificationCheckboxes.nth(i).check();
  }
  
  // Verify all 3 are checked
  for (let i = 0; i < Math.min(3, count); i++) {
    await expect(notificationCheckboxes.nth(i)).toBeChecked();
  }
  
  // TOGGLE: A specific checkbox
  const agreeCheckbox = page.getByLabel(/agree to terms/i);
  
  if (await agreeCheckbox.isChecked()) {
    await agreeCheckbox.uncheck();
  } else {
    await agreeCheckbox.check();
  }
  
  // CLICK: Submit
  await page.getByRole('button', { name: /submit/i }).click();
});

  await page.goto('https://example.com/shop');
  
  const products = page.locator('.product-card');
  const cart = page.locator('[data-testid="shopping-cart"]');
  
  // Get initial cart count
  let cartCount = 0;
  
  // DRAG: First 3 products to cart
  const productCount = await products.count();
  const itemsToDrag = Math.min(3, productCount);
  
  for (let i = 0; i < itemsToDrag; i++) {
    await products.nth(i).dragTo(cart);
    cartCount++;
    
    // Wait between drags
    await page.waitForTimeout(300);
  }
  
  // Verify items in cart
  const cartItems = cart.locator('.cart-item');
  expect(await cartItems.count()).toBe(cartCount);
});

test('load more items in infinite scroll', async ({ page }) => {
  await page.goto('https://example.com/infinite-feed');
  
  // Get initial items
  const items = page.locator('.feed-item');
  let initialCount = await items.count();
  console.log('Initial items:', initialCount);
  
  // SCROLL: To bottom
  const lastItem = items.last();
  await lastItem.scrollIntoViewIfNeeded();
  
  // Wait for loading indicator
  const loader = page.getByText(/loading more/i);
  await expect(loader).toBeVisible({ timeout: 5000 });
  
  // WAIT: For new items to load
  await page.waitForFunction(async () => {
    const currentCount = await page.locator('.feed-item').count();
    return currentCount > initialCount;
  });
  
  // Get new count
  const newCount = await items.count();
  console.log('Items after scroll:', newCount);
  expect(newCount).toBeGreaterThan(initialCount);
  
  // SCROLL: Again to load more
  const newLastItem = page.locator('.feed-item').last();
  await newLastItem.scrollIntoViewIfNeeded();
  
  // WAIT: For next batch
  await page.waitForFunction(async () => {
    const currentCount = await page.locator('.feed-item').count();
    return currentCount > newCount;
  });
});

SUMMARY TABLE: Which Action to Use

┌──────────────────┬──────────────────┬──────────────────────────┐
│ Scenario         │ Action Method    │ Example                  │
├──────────────────┼──────────────────┼──────────────────────────┤
│ Type in textbox  │ fill()           │ input.fill('text')       │
│ Click button     │ click()          │ button.click()           │
│ Select dropdown  │ selectOption()   │ select.selectOption('v') │
│ Check checkbox   │ check()          │ checkbox.check()         │
│ Type slowly      │ pressSequentially│ input.pressSequentially()│
│ Press key        │ press()          │ page.press('Enter')      │
│ Upload file      │ setInputFiles()  │ input.setInputFiles(f)   │
│ Set focus        │ focus()          │ input.focus()            │
│ Drag element     │ dragTo()         │ elem.dragTo(target)      │
│ Scroll element   │ scrollIntoView   │ elem.scrollIntoViewIfNeed│
└──────────────────┴──────────────────┴──────────────────────────┘

BEST PRACTICES CHECKLIST:

□ Use getByRole(), getByLabel() instead of selectors
□ Prefer fill() over pressSequentially()
□ Avoid { force: true }
□ Wait for actionability explicitly if needed
□ Use meaningful test descriptions
□ Separate test setup from verification
□ Handle dynamic content with waitForFunction()
□ Test actual user workflows
□ Verify success state, not just action completion
□ Keep tests focused and independent
